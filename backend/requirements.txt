# DropYourImage POC — backend dependencies
# Pinned deliberately: a 5-day sprint cannot absorb a surprise upstream break.
# Every line needs a reason. If you add one without a reason, remove it.

# --- determinism ---
threadpoolctl==3.5.0               # forces BLAS to genuinely 1 thread — see app/__init__.py

# --- imaging core (no AI, no network) ---
numpy==1.26.4                     # all pixel maths; every imaging stage is float32 arrays
                                  # held at 1.x: pytoshop (PSD fallback) predates numpy 2
opencv-python-headless==4.10.0.84 # contours for centring, morphology for the edge band
                                  # headless: no libGL on a server. Do NOT swap for opencv-python
Pillow==10.4.0                    # decode/encode, EXIF, ICC read+write via ImageCms (littleCMS)
rawpy==0.27.0                     # DECODING camera raw (CR2/CR3/NEF/CRW/DNG/.raw) via libraw.
                                  # Decode only — libraw ships no encoder and none exists, which
                                  # is why raw inputs are delivered as 16-bit TIFF. See
                                  # app/imaging/formats.py. Needs libraw at the system level;
                                  # the manylinux wheel bundles it.
tifffile==2026.3.3                # WRITING true 16-bit RGB TIFF. Pillow cannot: its RGB mode is
                                  # 8-bit and fromarray(uint16, mode="RGB") silently writes a
                                  # scrambled 8-bit file rather than raising. Was already present
                                  # as a psd-tools dependency; pinned here now that we import it
                                  # directly. Use compression="zlib" — its LZW needs imagecodecs.

# --- PSD ---
psd-tools==1.9.31                 # READING PSDs for the output validator. Not a writer
pytoshop==1.2.1                   # WRITING the layered PSD when Adobe is unavailable, which is
                                  # every deployment today — Firefly is enterprise-quoted, so this
                                  # is the only path that actually ships a PSD. Previously left
                                  # unpinned/uninstalled, which made the whole PSD stage fail.
                                  # Held at 1.2.1: app/psd/fallback.py works around two defects in
                                  # this exact wheel (broken RLE packbits, layer-name NUL), and it
                                  # is why numpy is held at 1.x. Builds from sdist, needs cython.

# --- API ---
fastapi==0.115.0
uvicorn[standard]==0.30.6
python-multipart==0.0.9           # multipart zip upload
pydantic==2.9.2                   # the frozen contract in app/models.py
pydantic-settings==2.5.2          # env-var config; keys never leave the server

# --- queue ---
rq==1.16.2
redis==5.0.8

# --- storage ---
boto3==1.35.24                    # S3 API; points at MinIO locally
google-cloud-storage==3.13.1      # GCS backend, selected by GCP_BUCKET_NAME. Alternative to S3,
                                  # not a replacement: both sit behind StorageBackend and the
                                  # deployment picks one. Needed for V4 signed download URLs,
                                  # which is the only way assets leave a private bucket.

# --- vendors ---
httpx==0.27.2                     # async HTTP to segmentation APIs

# --- security ---
python-magic==0.4.27              # MIME sniffing on zip entries; extensions are not trusted
                                  # needs libmagic1 (see scripts/setup.sh)

# --- tests ---
pytest==8.3.3
pytest-asyncio==0.24.0
